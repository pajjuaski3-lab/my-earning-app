val functions = FirebaseFunctions.getInstance() fun requestManualPayout( amountRupees: Long, method: String, destination: String, onResult: (String) -> Unit ) { val data = hashMapOf( "amountRupees" to amountRupees, "method" to method, "destination" to destination ) functions.getHttpsCallable("requestWithdrawal") .call(data) .addOnSuccessListener { result -> val response = result.data as Map<*, *> onResult("Request ID: ${response["requestId"]}") } .addOnFailureListener { error -> onResult("Request ವಿಫಲವಾಗಿದೆ: ${error.message}") } }
